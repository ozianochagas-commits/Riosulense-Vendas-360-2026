/** * RIOSULENSE — API do Dashboard de Vendas * Colunas da planilha (linha 1, exatamente nesta ordem): * mes | representante | meta | realizado | previsao | carteira * * mes usa o formato AAAA-MM (ex.: 2026-08) e deve estar * formatado como TEXTO SIMPLES na planilha. */ var HEADERS = ['mes', 'representante', 'meta', 'realizado', 'previsao', 'carteira']; function getSheet_() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; // Cria o cabeçalho se a planilha estiver vazia. if (sheet.getLastRow() === 0) { sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]); } return sheet; } function json_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function toNumber_(v) { if (typeof v === 'number') return isFinite(v) ? v : 0; var s = String(v === null || v === undefined ? '' : v).trim(); if (!s) return 0; s = s.replace(/R\$/gi, '').replace(/\s/g, ''); var comma = s.lastIndexOf(','); var dot = s.lastIndexOf('.'); if (comma > dot && comma !== -1) s = s.replace(/\./g, '').replace(',', '.'); else if (comma !== -1) s = s.replace(/\./g, '').replace(',', '.'); var n = Number(s); return isFinite(n) ? n : 0; } /** Normaliza o mês para AAAA-MM, aceitando texto ou data real. */ function toMonth_(v) { if (v instanceof Date) { return v.getFullYear() + '-' + ('0' + (v.getMonth() + 1)).slice(-2); } var s = String(v === null || v === undefined ? '' : v).trim(); var m = s.match(/^(\d{4})-(\d{1,2})/); if (m) return m[1] + '-' + ('0' + m[2]).slice(-2); return s; } // =============================================================== // LEITURA — GET // =============================================================== function doGet() { try { var sheet = getSheet_(); var lastRow = sheet.getLastRow(); if (lastRow < 2) return json_([]); var values = sheet.getRange(2, 1, lastRow - 1, HEADERS.length).getValues(); var rows = []; for (var i = 0; i < values.length; i++) { var r = values[i]; var mes = toMonth_(r[0]); var rep = String(r[1] || '').trim(); if (!mes || !rep) continue; // ignora linhas em branco rows.push({ mes: mes, representante: rep, meta: toNumber_(r[2]), realizado: toNumber_(r[3]), previsao: toNumber_(r[4]), carteira: toNumber_(r[5]) }); } return json_(rows); } catch (e) { return json_({ error: e.toString() }); } } // =============================================================== // GRAVAÇÃO — POST (upsert por mes + representante) // =============================================================== function doPost(e) { var lock = LockService.getScriptLock(); try { // Evita que dois salvamentos simultâneos criem linhas duplicadas. lock.waitLock(20000); if (!e || !e.postData || !e.postData.contents) { return json_({ ok: false, error: 'Requisição sem corpo (postData vazio).' }); } var body = JSON.parse(e.postData.contents); var mes = toMonth_(body.mes); var rep = String(body.representante || '').trim(); if (!mes || !rep) { return json_({ ok: false, error: 'Campos "mes" e "representante" são obrigatórios.' }); } var novaLinha = [ mes, rep, toNumber_(body.meta), toNumber_(body.realizado), toNumber_(body.previsao), toNumber_(body.carteira) ]; var sheet = getSheet_(); var lastRow = sheet.getLastRow(); // Procura a linha existente com o mesmo mes + representante. if (lastRow >= 2) { var chaves = sheet.getRange(2, 1, lastRow - 1, 2).getValues(); for (var i = 0; i < chaves.length; i++) { if (toMonth_(chaves[i][0]) === mes && String(chaves[i][1] || '').trim() === rep) { sheet.getRange(i + 2, 1, 1, HEADERS.length).setValues([novaLinha]); SpreadsheetApp.flush(); return json_({ ok: true, acao: 'atualizado', linha: i + 2 }); } } } // Não existe: cria no final. sheet.appendRow(novaLinha); SpreadsheetApp.flush(); return json_({ ok: true, acao: 'criado', linha: sheet.getLastRow() }); } catch (err) { return json_({ ok: false, error: err.toString() }); } finally { try { lock.releaseLock(); } catch (ignore) {} } }
